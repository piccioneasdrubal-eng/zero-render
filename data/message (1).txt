  'use strict';

  const SCRIPT_DELAY = 6000;
  const POLL_INTERVAL = 100000;

  // --- WebSocket proxy for server switches -----------------
  const _OrigWebSocket = window.WebSocket;

            tex.customSkinMap.clear();

          }
        } catch (e) { /* ignore */ }
        return new target(...args);
      }
    });
    try { window.WebSocket = ProxiedWebSocket; } catch (e) {}
  }

  setTimeout(init, SCRIPT_DELAY);

  let _texObj = null;
  let _webpackReq = null;

  function looksLikeTex(v) {
    if (!v || typeof v !== 'object') return false;
    try { return v.customSkinMap instanceof Map; } catch (e) { return false; }
  }

  function getWebpackRequire() {
    if (_webpackReq && _webpackReq.m) return _webpackReq;
    const chunkKeyCandidates = Object.keys(window).filter(k => /^webpackChunk/.test(k));
    for (const key of chunkKeyCandidates) {
      try {
        const arr = window[key];
        if (!arr || typeof arr.push !== 'function') continue;
        let captured = null;
        arr.push([[Symbol('probe')], {}, (req) => { captured = req; }]);
        if (captured && captured.m) { _webpackReq = captured; return captured; }
      } catch (e) {}
    }
    return null;
  }

  function scanWebpackModulesForTex() {
    const req = getWebpackRequire();
    if (!req || !req.m) return null;
    const ids = Object.keys(req.m);
    for (const id of ids) {
      let exp;
      try { exp = req(id); } catch (e) { continue; }
      if (!exp) continue;
      if (looksLikeTex(exp)) return exp;
      if (typeof exp === 'object') {
        let sub = [];
        try { sub = Object.keys(exp); } catch (e) { continue; }
        if (sub.length > 30) continue;
        for (const k of sub) {
          let v;
          try { v = exp[k]; } catch (e) { continue; }
          if (looksLikeTex(v)) return v;
        }
      }
    }
    return null;
  }

  function findTex() {
    if (_texObj && looksLikeTex(_texObj)) return _texObj;
    if (looksLikeTex(window.Texture)) { _texObj = window.Texture; window.__texObjRef = _texObj; return window.Texture; }
    let keys = [];
    try { keys = Object.keys(window); } catch (e) {}
    for (const k of keys) {
      let v;
      try { v = window[k]; } catch (e) { continue; }
      if (looksLikeTex(v)) { _texObj = v; window.__texObjRef = _texObj; return v; }
    }
    for (const k of keys) {
      let v;
      try { v = window[k]; } catch (e) { continue; }
      if (!v || typeof v !== 'object') continue;
      if (v instanceof Node || v instanceof Window) continue;
      let sub = [];
      try { sub = Object.keys(v); } catch (e) { continue; }
      if (sub.length > 50) continue;
      for (const k2 of sub) {
        let v2;
        try { v2 = v[k2]; } catch (e) { continue; }
        if (looksLikeTex(v2)) { _texObj = v2; window.__texObjRef = _texObj; return v2; }
      }
    }
    const fromWebpack = scanWebpackModulesForTex();
    if (fromWebpack) { _texObj = fromWebpack; window.__texObjRef = _texObj; return fromWebpack; }
    return null;
  }

  window.__texClear = function () {
    const t = findTex();
    if (t && t.customSkinMap && typeof t.customSkinMap.clear === 'function') {
      t.customSkinMap.clear();
      persistentPlayerMap.clear();
      console.log('[Tracker] manually cleared skin data');
      return true;
    }
    persistentPlayerMap.clear();
    return false;
  };

  function stripTrailingDigits(str) {
    return String(str).replace(/\d+$/, '').trim();
  }

  function fixImageUrl(url) {
    if (!url) return url;
    url = String(url);
    if (url.includes('imgur.com/') && !url.includes('i.imgur.com')) {
      const match = url.match(/imgur\.com\/(?:gallery\/)?([a-zA-Z0-9]+)(?:\.[a-z]+)?$/);
      if (match) { return `https://i.imgur.com/${match[1]}.png`; }
    }
    if (url.includes('i.imgur.com') && url.startsWith('http:')) {
      return url.replace('http:', 'https:');
    }
    if (url.startsWith('http://')) {
      return url.replace('http://', 'https://');
    }
    return url;
  }

  // --- Active Player Scanner with Permanent Retention ---------------------
  function getActivePlayers() {
    let cells = null;
    try {
      if (window.cells && Array.isArray(window.cells)) {
        cells = window.cells;
      } else if (window.application && typeof window.application.getCells === 'function') {
        cells = window.application.getCells();
      } else if (window.core && Array.isArray(window.core.cells)) {
        cells = window.core.cells;
      } else if (window.agar && Array.isArray(window.agar.allCells)) {
        cells = window.agar.allCells;
      }
    } catch (e) { /* safe fallback */ }

    const tex = findTex();
    const csm = tex && tex.customSkinMap;

    if (cells && Array.isArray(cells) && cells.length > 0) {
      cells.forEach(c => {
        if (!c) return;
        const name = c.name || c.n || c.nick;
        if (!name) return;
        if (c.isVirus || c.virus || c.isFood || c.food) return;

        const size = c.size || c.s || 0;
        const mass = c.mass || c.m || (size > 0 ? Math.floor((size * size) / 100) : 0);
        const x = c.x || 0;
        const y = c.y || 0;

        let skinURL = c.skinURL || c.skin || c.customSkin || '';
        if (!skinURL && csm && csm.has(name)) {
          const inner = csm.get(name);
          if (inner instanceof Map) {
            inner.forEach(u => { if (!skinURL) skinURL = u; });
          }
        }
        skinURL = fixImageUrl(skinURL);

        if (!persistentPlayerMap.has(name)) {
          persistentPlayerMap.set(name, {
            id: name,
            name: stripTrailingDigits(name) || name,
            skinURL: skinURL,
            totalMass: mass,
            sumX: x,
            sumY: y,
            cellCount: 1,
            lastMass: mass,
            lastX: x,
            lastY: y
          });
        } else {
          const p = persistentPlayerMap.get(name);
          p.totalMass += mass;
          p.sumX += x;
          p.sumY += y;
          p.cellCount += 1;
          if (!p.skinURL && skinURL) p.skinURL = skinURL;
        }
      });
    }

    if (csm && typeof csm.forEach === 'function') {
      try {
        csm.forEach((innerMap, playerName) => {
          if (!(innerMap instanceof Map)) return;
          let skinURL = '';
          innerMap.forEach(url => { if (!skinURL) skinURL = fixImageUrl(url); });
          if (!skinURL) return;

          if (!persistentPlayerMap.has(playerName)) {
            persistentPlayerMap.set(playerName, {
              id: playerName,
              name: stripTrailingDigits(playerName) || playerName,
              skinURL: skinURL,
              totalMass: 0,
              sumX: 0,
              sumY: 0,
              cellCount: 0,
              lastMass: '???',
              lastX: '?',
              lastY: '?'
            });
          } else {
            const p = persistentPlayerMap.get(playerName);
            if (!p.skinURL && skinURL) p.skinURL = skinURL;
          }
        });
      } catch (e) { /* safe iteration */ }
    }

    const players = [];
    persistentPlayerMap.forEach((p, key) => {
      let finalMass;
      let avgX;
      let avgY;

      if (p.cellCount > 0) {
        avgX = Math.round(p.sumX / p.cellCount);
        avgY = Math.round(p.sumY / p.cellCount);
        finalMass = p.totalMass;

        p.lastMass = finalMass;
        p.lastX = avgX;
        p.lastY = avgY;
      } else {
        finalMass = p.lastMass !== undefined ? p.lastMass : '???';
        avgX = p.lastX !== undefined ? p.lastX : '?';
        avgY = p.lastY !== undefined ? p.lastY : '?';
      }

      players.push({
        id: p.id,
        name: p.name,
        skinURL: p.skinURL || '',
        mass: finalMass,
        x: avgX,
        y: avgY
      });

      p.totalMass = 0;
      p.sumX = 0;
      p.sumY = 0;
      p.cellCount = 0;
    });

    players.sort((a, b) => {
      const mA = typeof a.mass === 'number' ? a.mass : -1;
      const mB = typeof b.mass === 'number' ? b.mass : -1;
      return mB - mA;
    });

    return players;
  }

  // --- UI Setup & New Features Integration --------------------------------
  function init() {
    console.log('Player Tracker initialized with counter, minimize toggle, and click-to-copy.');
    const overlay = createOverlay();
    const listEl = overlay.querySelector('#player-tracker-list');
    const searchInput = overlay.querySelector('#player-search-input');
    const countBadge = overlay.querySelector('#player-count-badge');
    const minimizeBtn = overlay.querySelector('#player-minimize-btn');

    minimizeBtn.addEventListener('click', () => {
      isMinimized = !isMinimized;
      listEl.style.display = isMinimized ? 'none' : 'flex';
      minimizeBtn.textContent = isMinimized ? '+' : '−';
    });

    searchInput.addEventListener('input', (e) => {
      searchTerm = e.target.value.toLowerCase().trim();
      update();
    });

    function update() {
      let players = getActivePlayers();
      countBadge.textContent = players.length;

      if (searchTerm) {
        players = players.filter(pl => pl.name.toLowerCase().includes(searchTerm) || pl.id.toLowerCase().includes(searchTerm));
      }

      renderList(listEl, players);
    }

    update();
    setInterval(update, POLL_INTERVAL);
  }

  function renderList(container, players) {
    if (!container) return;
    container.innerHTML = '';

    if (!players.length) {
      const empty = document.createElement('div');
      empty.textContent = 'No players found';
      empty.style.fontSize = '12px';
      empty.style.opacity = '0.7';
      empty.style.textAlign = 'center';
      empty.style.padding = '8px 0';
      container.appendChild(empty);
      return;
    }

    players.forEach((pl) => {
      const item = document.createElement('div');
      item.title = `Click to copy name: ${pl.name}`;
      item.style.cssText = `
        display: flex;
        flex-direction: row;
        align-items: center;
        justify-content: space-between;
        padding: 3px 4px;
        border-radius: 4px;
        font-size: 11px;

        gap: 6px;
        background: rgba(255, 255, 255, 0.03);
        cursor: pointer;
        transition: background 0.2s, border-color 0.2s;
        border: 1px solid transparent;
      `;
      item.addEventListener('mouseenter', () => { item.style.background = 'rgba(255, 255, 255, 0.08)'; });
      item.addEventListener('mouseleave', () => { item.style.background = 'rgba(255, 255, 255, 0.03)'; });
