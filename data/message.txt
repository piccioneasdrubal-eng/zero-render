  if (typeof _OrigWebSocket === 'function') {

  function looksLikeTex(v) {
    if (!v || typeof v !== 'object') return false;

  }

  function getWebpackRequire() {
    if (_webpackReq && _webpackReq.m) return _webpackReq;
    const chunkKeyCandidates = Object.keys(window).filter(k => /^webpackChunk/.test(k));
    for (const key of chunkKeyCandidates) {
      try {
        const arr = window[key];
        if (!arr || typeof arr.push !== 'function') continue;

        // Create a unique probe entry so we can always remove it afterwards.
        const probeRuntime = (req) => { captured = req; };
        const probe = [[Symbol('probe')], {}, probeRuntime];
        let captured = null;

        try {
          arr.push(probe);
          // pushing the probe will cause webpack to call the runtime (where we capture)
          // If captured is populated, it exposes 'm' (modules)
          if (captured && captured.m) { _webpackReq = captured; return captured; }
        } catch (e) {
          // ignore failures while pushing probe
        } finally {
          // Remove our probe entry to avoid leaving spurious data in the chunk array.
          try {
            const idx = arr.indexOf(probe);
            if (idx !== -1) arr.splice(idx, 1);
          } catch (er) { /* ignore cleanup errors */ }
        }
      } catch (e) {}
    }
    return null;
  }

  function scanWebpackModulesForTex() {
    const req = getWebpackRequire();
    if (!req || !req.m) return null;
    const ids = Object.keys(req.m);

      try { exp = req(id); } catch (e) { continue; }
      if (!exp) continue;
      if (looksLikeTex(exp)) return exp;
      if (typeof exp === 'object') {
        let sub = [];
        try { sub = Object.keys(exp); } catch (e) { continue; }
        if (sub.length > 30) continue;
        for (const k of sub) {
          let v;
          try { v = exp[k]; } catch (e) { continue; }
          if (looksLikeTex(v)) return v;
        }
      }
    }
    return null;
  }

  function findTex() {
    if (_texObj && looksLikeTex(_texObj)) return _texObj;

  window.__texClear = function () {

  function stripTrailingDigits(str) {
    return String(str).replace(/\d+$/, '').trim();
  }

  function fixImageUrl(url) {
    if (!url) return url;
    url = String(url).trim();
    // Strip query/hash
    url = url.split(/[?#]/)[0];

    // Support protocol-relative URLs
    if (url.startsWith('//')) url = 'https:' + url;

    // Ensure https
    if (url.startsWith('http://')) url = url.replace('http://', 'https://');

    // Imgur normalization: support i.imgur.com, imgur.com/{id}, imgur.com/a/{album}, gallery
    try {
      const hostMatch = url.match(/(?:https?:\/\/)?(?:i\.)?imgur\.com\/(.+)$/i);
      if (hostMatch) {
        const tail = hostMatch[1].replace(/\/+$/, '');
        // If it's an album/gallery, try to extract last segment (album id won't map to direct image but we try)
        const simpleIdMatch = tail.match(/(?:a\/|gallery\/)?([a-zA-Z0-9]+)(?:\.[a-z]+)?$/);
        if (simpleIdMatch) {
          return `https://i.imgur.com/${simpleIdMatch[1]}.png`;
        }
      }
    } catch (e) { /* ignore URL parsing errors */ }

    // If still starts with http(s) or data: leave as-is (after https normalization)
    return url;
  }

  // --- Active Player Scanner with Status & Retention ----------------------
  function getActivePlayers() {
    let cells = null;
    try {
      if (window.cells && Array.isArray(window.cells)) {
        cells = window.cells;
      } else if (window.application && typeof window.application.getCells === 'function') {
        cells = window.application.getCells();
      } else if (window.core && Array.isArray(window.core.cells)) {
        cells = window.core.cells;
      } else if (window.agar && Array.isArray(window.agar.allCells)) {
        cells = window.agar.allCells;
      }
    } catch (e) { /* safe fallback */ }

    const tex = findTex();
    const csm = tex && tex.customSkinMap;

    if (cells && Array.isArray(cells) && cells.length > 0) {
      cells.forEach(c => {
        if (!c) return;
        const name = c.name || c.n || c.nick;
        if (!name) return;
        if (c.isVirus || c.virus || c.isFood || c.food) return;

        const size = c.size || c.s || 0;
        const mass = c.mass || c.m || (size > 0 ? Math.floor((size * size) / 100) : 0);
        const x = c.x || 0;
        const y = c.y || 0;

        let skinURL = c.skinURL || c.skin || c.customSkin || '';
        if (!skinURL && csm && csm.has(name)) {
          const inner = csm.get(name);
          if (inner instanceof Map) {
            inner.forEach(u => { if (!skinURL) skinURL = u; });
          }
        }
        skinURL = fixImageUrl(skinURL);

        if (!persistentPlayerMap.has(name)) {
          persistentPlayerMap.set(name, {
            id: name,
            name: stripTrailingDigits(name) || name,
            skinURL: skinURL,
            totalMass: mass,
            sumX: x,
            sumY: y,
            cellCount: 1,
            lastMass: mass,
            lastX: x,
            lastY: y,
            peakMass: mass
          });
        } else {
          const p = persistentPlayerMap.get(name);
          p.totalMass += mass;
          p.sumX += x;
          p.sumY += y;
          p.cellCount += 1;
          if (!p.skinURL && skinURL) p.skinURL = skinURL;
          if (mass > p.peakMass) p.peakMass = mass;
        }
      });
    }

    if (csm && typeof csm.forEach === 'function') {
      try {
        csm.forEach((innerMap, playerName) => {
          if (!(innerMap instanceof Map)) return;
          let skinURL = '';
          innerMap.forEach(url => { if (!skinURL) skinURL = fixImageUrl(url); });
          if (!skinURL) return;

          if (!persistentPlayerMap.has(playerName)) {
            persistentPlayerMap.set(playerName, {
              id: playerName,
              name: stripTrailingDigits(playerName) || playerName,
              skinURL: skinURL,
              totalMass: 0,
              sumX: 0,
              sumY: 0,
              cellCount: 0,
              lastMass: '???',
              lastX: '?',
              lastY: '?',
              peakMass: 0
            });
          } else {
            const p = persistentPlayerMap.get(playerName);
            if (!p.skinURL && skinURL) p.skinURL = skinURL;
          }
        });
      } catch (e) { /* safe iteration */ }
    }

    const players = [];
    persistentPlayerMap.forEach((p) => {
      let finalMass;
      let avgX;
      let avgY;
      let isActiveNow = (p.cellCount > 0);

      if (isActiveNow) {
        avgX = Math.round(p.sumX / p.cellCount);
        avgY = Math.round(p.sumY / p.cellCount);
        finalMass = p.totalMass;

        p.lastMass = finalMass;
        p.lastX = avgX;
        p.lastY = avgY;
        if (finalMass > p.peakMass) p.peakMass = finalMass;
      } else {
        finalMass = p.lastMass !== undefined ? p.lastMass : '???';
        avgX = p.lastX !== undefined ? p.lastX : '?';
        avgY = p.lastY !== undefined ? p.lastY : '?';
      }

      players.push({
        id: p.id,
        name: p.name,
        skinURL: p.skinURL || '',
        mass: finalMass,
        peakMass: p.peakMass,
        x: avgX,
        y: avgY,
        isActiveNow: isActiveNow
      });

      p.totalMass = 0;
      p.sumX = 0;
      p.sumY = 0;
      p.cellCount = 0;
    });

    saveTrackerToStorage();

    players.sort((a, b) => {
      if (sortMode === 'name') {
        return a.name.localeCompare(b.name);
      } else if (sortMode === 'peak') {
        const pA = typeof a.peakMass === 'number' ? a.peakMass : -1;
        const pB = typeof b.peakMass === 'number' ? b.peakMass : -1;
        return pB - pA;
      } else {
        const mA = typeof a.mass === 'number' ? a.mass : -1;
        const mB = typeof b.mass === 'number' ? b.mass : -1;
        return mB - mA;
      }
    });

    return players;
  }

  function exportTrackerData() {
    const players = [];
    persistentPlayerMap.forEach((p) => {
      players.push({
        id: p.id,
        name: p.name,
        skinURL: p.skinURL,
        lastMass: p.lastMass,
        peakMass: p.peakMass,
        lastX: p.lastX,
        lastY: p.lastY
      });
    });

    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(players, null, 2));
    const downloadAnchor = document.createElement('a');
    downloadAnchor.setAttribute("href", dataStr);
    downloadAnchor.setAttribute("download", `agar_players_${Date.now()}.json`);
    document.body.appendChild(downloadAnchor);
    downloadAnchor.click();
    downloadAnchor.remove();
  }

      searchTerm = e.target.value.toLowerCase().trim();

      sortMode = e.target.value;
      update();
    });

    exportBtn.addEventListener('click', () => {
      exportTrackerData();
    });

      let players = getActivePlayers();
      const activeCount = players.filter(pl => pl.isActiveNow).length;

      if (showOnlineOnly) {
        players = players.filter(pl => pl.isActiveNow);
      }

      if (searchTerm) {
        players = players.filter(pl => pl.name.toLowerCase().includes(searchTerm) || pl.id.toLowerCase().includes(searchTerm));
      }

      currentRenderedPlayers = players;
      countBadge.textContent = `${activeCount} / ${persistentPlayerMap.size}`;

      renderList(listEl, players);
    }

    update();
    setInterval(update, POLL_INTERVAL);
  }

      `;
